import com.familyexpense.tracker.sms.*

private var pass = 0
private var fail = 0
private val failures = mutableListOf<String>()

private fun check(label: String, cond: Boolean, detail: String = "") {
    if (cond) pass++ else { fail++; failures += "$label  $detail" }
}

private fun expectDebit(label: String, sender: String, body: String, rupees: Double, merchant: String? = null) {
    when (val r = SmsParser.parse(sender, body, 1_700_000_000_000L)) {
        is ParseOutcome.Parsed -> {
            val t = r.transaction
            check(label, t.direction == Direction.DEBIT && t.amountRupees == rupees,
                "-> ${t.direction} ${t.amountRupees} (want DEBIT $rupees) [${t.matchedBy}] merchant=${t.merchant}")
            if (merchant != null) check("$label/merchant", t.merchant == merchant,
                "-> '${t.merchant}' want '$merchant' [${t.matchedBy}]")
        }
        else -> check(label, false, "-> $r")
    }
}

private fun expectCredit(label: String, sender: String, body: String, rupees: Double) {
    when (val r = SmsParser.parse(sender, body, 1_700_000_000_000L)) {
        is ParseOutcome.Parsed -> {
            val t = r.transaction
            check(label, t.direction == Direction.CREDIT && t.amountRupees == rupees,
                "-> ${t.direction} ${t.amountRupees} (want CREDIT $rupees) [${t.matchedBy}]")
        }
        else -> check(label, false, "-> $r")
    }
}

private fun expectRejected(label: String, sender: String, body: String) {
    when (val r = SmsParser.parse(sender, body, 1_700_000_000_000L)) {
        is ParseOutcome.Parsed -> check(label, false,
            "-> WRONGLY PARSED as ${r.transaction.direction} ${r.transaction.amountRupees} [${r.transaction.matchedBy}]")
        else -> { pass++ }
    }
}

fun main() {
    // ============ DEBITS (verbatim shapes from real corpora) ============

    // The highest-volume debit SMS in India. No currency token, single decimal.
    expectDebit("sbi_upi", "VM-SBIUPI-S",
        "Dear UPI user A/C X1234 debited by 150.0 on date 05Mar24 trf to SWIGGY Refno 406512345678. If not u? call 1800111109. -SBI",
        150.0, "Swiggy")

    expectDebit("hdfc_sent", "VM-HDFCBK-S",
        "Sent Rs.100.00\nFrom HDFC Bank A/C *0000\nTo CUSTOMER NAME\nOn 17/05/26\nRef 000000000000",
        100.0)

    // Merchant literally contains "CREDIT" -- a keyword scan books this as income.
    expectDebit("hdfc_card_pzcredit", "VM-HDFCDC-S",
        "Spent Rs.3000 From HDFC Bank Card x0000 At PZCREDIT0000000 On 2026-05-02:00:17:56 Bal Rs.142.26",
        3000.0, "PayZapp")

    // Contains "credited" for the counterparty but is a debit.
    expectDebit("icici_upi", "JD-ICICIT-S",
        "ICICI Bank Acct XX000 debited for Rs 14.00 on 09-May-26; Pune Metro credited. UPI:000000000000. Call 18002662 for dispute.",
        14.0, "Pune Metro")

    expectDebit("axis_p2m", "AD-AXISBK-S",
        "INR 726.00 debited A/c no. XX1234 05-09-26, 10:19:49 UPI/P2M/000000000000/RAHUL SHARMA Not you? SMS BLOCKUPI Cust ID to 919900000000 Axis Bank",
        726.0)

    // "Dr." abbreviation -- missed by parsers that only look for "debited".
    expectDebit("bob_dr", "VM-BOBTXN-S",
        "Rs.230.00 Dr. from A/C XXXXXX1234 and Cr. to example@okbank. Ref:000000000000. AvlBal:Rs618.85(2026:09:06 09:56:33). Not you? Call 18005700/5000-BOB",
        230.0)

    // "Rs:" with a colon, payee after "Fvg:".
    expectDebit("union_colon", "VM-UNIONB-S",
        "Union Bank of India A/c *0000 Debited Rs:151.00 on 25-08-2026 09:50:01 by Mob Bk ref no 000000000000, Fvg: RAHUL SHARMA Avl Bal Rs:7138.46.",
        151.0)

    // Lakh grouping.
    expectDebit("lakh_grouping", "VM-HDFCBK-S",
        "Rs.1,23,456.78 spent on HDFC Bank Card x0000 at BIG STORE on 2026-07-12:17:00:56.Not U?",
        123456.78)

    // Real ICICI card spend that mentions EMI twice -- must NOT be excluded.
    expectDebit("icici_card_emi_words", "JD-ICICIT-S",
        "Rs 100.00 spent on ICICI Bank Card XX0000 on 16-May-26 at SAMPLE MERCHANT. Avl Lmt: Rs 200.00. To convert this txn to EMI give a missed call on 9924667667. Know more about EMI conversion.",
        100.0)

    expectDebit("sbi_card", "VM-SBICRD-S",
        "Rs.259.00 spent on your SBI Credit Card ending with 1234 on 15Jan26. Your available limit is Rs.1,235.00.",
        259.0)

    expectDebit("amex", "VM-AMEXIN-S",
        "Alert: You've spent INR 4,411.00 on your AMEX Corp Card ** 21006 at CWT INDIA on 8 January 2026 at 11:29 PM IST.",
        4411.0)

    expectDebit("kotak_sent", "VM-KBANKT-S",
        "Sent Rs.500.00 from Kotak Bank A/c X0000 to SampleCo Services Ind on 10-09-26. UPI Ref 000000000000.",
        500.0)

    expectDebit("federal_atm", "VM-FEDBNK-S",
        "Rs 1500 withdrawn@ YBL CHAN on 21JAN26 17:59 Bal Rs 7517.94 Ref 602117126490.",
        1500.0)

    // ============ CREDITS ============

    expectCredit("hdfc_credit", "VM-HDFCBK-S",
        "Credit Alert!\nRs.1.00 credited to HDFC Bank A/c XX0000 on 09-05-26\nfrom VPA customer@bank (UPI 000000000000)",
        1.0)

    expectCredit("sbi_credit", "AD-SBIINB-S",
        "Dear SBI User, your A/c X0000-credited by Rs.12345 on 28Jul26 transfer from Sample Name Ref No 123456789012 -SBI",
        12345.0)

    expectCredit("refund", "JD-ICICIT-S",
        "Alert! Rs. 262.56 refunded by SampleMerchant Payments BANGALORE IND on 17/MAY/2026 & adjusted against HDFC Bank Credit Card 0000",
        262.56)

    // ============ MUST BE REJECTED ============

    // Carries an amount AND a merchant -- the nastiest false positive.
    expectRejected("otp_with_amount", "VM-HDFCBK-S",
        "482913 is OTP for txn of INR 2,499.00 at AMAZON on HDFC Bank card ending 4411. Valid till 10:42. Do not share OTP")

    expectRejected("emandate_future", "VM-HDFCBK-S",
        "E-Mandate!\nRs.3821.00 will be deducted on 02/03/26, 00:00:00\nFor Rentomojo mandate\nUMN 6cc417@ybl\nMaintain Balance\n-HDFC Bank")

    expectRejected("failed_txn", "VM-CANBNK-S",
        "Dear Customer, txn of Rs.637.70 thru A/C XX1234 on 18-8-26 at 14:16:16 to ACME STORE failed due to INSUFFICIENT FUNDS-Canara Bank")

    expectRejected("declined", "VM-RBLBNK-S",
        "ALERT: Credit limit exceeded. Your transaction for INR 1,793.00 at ROLLA HYPER MARKET was declined as you have exceeded the limit")

    expectRejected("statement", "VM-BOBCRD-S",
        "Hi, Your April-2026 bill of Rs.7,777.77 is ready. Please pay by 05 May, 2026 through the OneCard app.")

    expectRejected("due_reminder", "VM-YESBNK-S",
        "Payment of Equitas Credit Card 0000 is due on 10/07/26. Min due Rs 1234.56 Total due Rs 12345.67.")

    expectRejected("asba", "VM-IDFCFB-S",
        "Your ASBA application for SAMPLEIPO is received and Application value of Rs 14999 is blocked in your registered Bank account on 14/08/2026.")

    expectRejected("future_refund", "JD-ICICIT-S",
        "Refund of Rs 1,299 for your order 403-1234567 has been processed and will be credited to your ICICI Bank card in 3-5 days.")

    // A bank alert from a mobile number is a scam. Highest-value fraud rule.
    expectRejected("scam_from_phone", "+919876543210",
        "Rs.5000.00 debited from your account. Call 9876543210 immediately if not you.")

    expectRejected("unknown_sender", "VM-SHOPPY-P",
        "Rs.500 spent at OUR STORE. Thanks for shopping!")

    // Reversal genuinely credited money even though it says "failed".
    expectCredit("reversal_despite_failed", "AD-SBIINB-S",
        "Rs 500.00 reversed and credited back to your A/c XX5678 on 23-09-26 for failed UPI txn Ref 626512349999. -SBI",
        500.0)

    // ============ amount parsing units ============
    check("paise_single_decimal", SmsParser.parseAmountToPaise("150.0") == 15000L)
    check("paise_lakh", SmsParser.parseAmountToPaise("1,23,456.78") == 12345678L)
    check("paise_western", SmsParser.parseAmountToPaise("211,476.24") == 21147624L)
    check("paise_no_decimal", SmsParser.parseAmountToPaise("3000") == 300000L)
    check("paise_leading_dot", SmsParser.parseAmountToPaise(".28") == 28L)

    println("\n  passed: $pass    failed: $fail")
    if (failures.isNotEmpty()) {
        println("\n  FAILURES:")
        failures.forEach { println("   - $it") }
    }
    if (fail > 0) System.exit(1)
}
